# 327: TAA + NIS'te hiç hareket etmeden nişan alıp bırakınca piramidin küplerinin arası birkaç saniye hafif yanıp sönüyordu. FPS betiği
# nişan geçişini üstel yaklaşımla yapar (aim01 hedefe asimptotik): FOV 50+ kare boyunca piksel altı değişiyor, kamera 1e-6'lık matris
# karşılaştırmasıyla "hareketli" sayılıyor, zamansal yöntemler bu sürede hareket yoluyla çiziyor (TAAU'nun dururken birikimi ve titreme
# tutucu kapalı). Kamera artık ekrandaki kayma 0.05 pikselin altındaysa duruyor sayılır. Nişan bırakıldıktan 60-120 kare sonra piramitte
# 8 seviyeden fazla değişen piksel: eski 1.45 %, şimdi 0.39 % (nişandan önce 0.007 %; yerel TAA 0.15 %)
# 327 (2. kısım): şablonun nişan geçişi üstel yaklaşımla 0.0002'ye kadar ~0.7 saniye sürüyordu; son ~0.4 saniyede görüş açısı ekranda
# 0.05-0.5 piksel kaydığı için zamansal AA bırakıştan sonra küplerin aralarında parıltı bırakıyordu. Geçiş artık en az doğrusal adımla
# sonlu sürede (~0.3 s) biter. Bırakıştan 30-70 kare sonra 8 seviyeden çok değişen: eski betik %2.10 (16'dan çok %0.31), yeni %0.11
# (%0.004); yerel TAA da aynı (bırakıştan 30-60 kare sonra %2.30 -> %0.003)
timeout 400
wait 60
focus Game
play
wait 30
entity move Player 0 0.9 9
wait 4
script set Player@FirstPersonShooter yaw float 0
script set Player@FirstPersonShooter pitch float -3
# Kapı aralığından görünen kayan hedefler ölçülen bölgeye girip çıkabiliyor (ölçüm o anda nerede olduklarına bağlıydı: durgun pencere
# %0.004 ile %0.27 arasında oynuyordu); ölçü yalnız piramidin aralıkları olsun diye durdurulur
script set "Moving Target 0@ShootingTarget" moving bool false
script set "Moving Target 1@ShootingTarget" moving bool false
script set "Moving Target 2@ShootingTarget" moving bool false
cvar r.AntiAliasing 3
cvar r.Upscaler 3
wait 400
game_flicker 60 0.30 0.30 0.56 0.74 max8% 0.05
wait 64
script set Player@FirstPersonShooter aimInput bool true
wait 110
script set Player@FirstPersonShooter aimInput bool false
wait 30
game_flicker 40 0.30 0.30 0.56 0.74 max8% 0.3
wait 64
stop
quit
