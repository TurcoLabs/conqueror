# Lighting > Ambient Occlusion > Full Resolution (Godot / Unity HDRP gibi): açıkken SSAO tam çözünürlükte (326: keskin, kenar dışına hale
# taşmaz; varsayılan), kapalıyken yarım çözünürlükte (yarı maliyet, aynı bulanıklaştırma ve derinliğe duyarlı büyütme). Ayar sahneyle
# kaydedilir, r.SSAOFullRes ve ConquerorScript Post.AOFullResolution ile değişir. Ölçü ao_view_sharp'taki gibi bölgedeki ayrıntı
wait_until 3000 layout_settled
wait 10
focus Viewport
cvar r.AntiAliasing 0
shading ao
camera 45 6 0 0 3 0
wait 60
viewport_region 0.375 0.375 0.5 0.5 lum detail >1.75
cvar r.SSAOFullRes 0
wait 60
viewport_region 0.375 0.375 0.5 0.5 lum detail <1.75
cvar r.SSAOFullRes 1
wait 60
viewport_region 0.375 0.375 0.5 0.5 lum detail >1.75
shading shaded
quit
